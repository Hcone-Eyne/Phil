# CELL 2: Mount Google Drive (with retry)
# If mount fails, click the link and allow access

from google.colab import drive
import os

# Try mounting with retry
MAX_RETRIES = 3
for attempt in range(1, MAX_RETRIES + 1):
    try:
        print(f"Mount attempt {attempt}/{MAX_RETRIES}...")
        drive.mount('/content/drive', force_remount=False)
        print("Drive mounted successfully!")
        break
    except Exception as e:
        print(f"Mount failed: {e}")
        if attempt < MAX_RETRIES:
            print("Retrying in 3 seconds...")
            import time
            time.sleep(3)
        else:
            print("\n=== ALTERNATIVE: Skip Drive, save locally ===")
            print("Checkpoints will save to /content/ (lost on disconnect)")
            print("But you can download before session ends!\n")
