# -- Combined convert + format function --
BATCH_SIZE = 50

def convert_and_format(batch_dict):
    """
    Convert CadQuery -> FreeCAD JSON -> formatted chat text.
    Returns list of (formatted_text, is_valid) tuples.
    """
    results = []
    codes = batch_dict.get("cadquery_file", [])

    for idx, code_bytes in enumerate(codes):
        try:
            # Decode bytes to string
            if isinstance(code_bytes, bytes):
                code = code_bytes.decode("utf-8", errors="ignore")
            else:
                code = str(code_bytes)

            if not code or not code.strip():
                continue

            # Convert
            spec = convert_cadquery(code)

            if spec is None:
                continue

            if not validate_json_spec(spec):
                continue

            # Format as chat
            response = json.dumps(spec, indent=2)
            messages = [
                {"role": "system", "content": SYSTEM_PROMPT},
                {"role": "user", "content": "Create a 3D model based on the provided CadQuery reference."},
                {"role": "assistant", "content": response},
            ]

            text = tokenizer.apply_chat_template(
                messages,
                tokenize=False,
                add_generation_prompt=False,
            )

            if text and len(text) > 10:
                results.append(text)

        except Exception as e:
            print(f"  Error processing sample: {e}")
            traceback.print_exc()
            continue

    return results

# -- Load checkpoint if exists --
converted_texts = []
dataset_idx = 0

if CONVERT_CHECKPOINT.exists():
    print(f"Found checkpoint at {CONVERT_CHECKPOINT}")
    with open(CONVERT_CHECKPOINT, "r") as f:
        checkpoint_data = json.load(f)
    converted_texts = checkpoint_data.get("texts", [])
    dataset_idx = checkpoint_data.get("dataset_idx", 0)
    print(f"Resuming from dataset index {dataset_idx}, {len(converted_texts)} valid samples so far")

# -- Convert with checkpointing --
print(f"\n=== Converting samples {dataset_idx} to {len(hf_dataset)} ===")
SAVE_EVERY = 200

for i in range(dataset_idx, len(hf_dataset), BATCH_SIZE):
    end_idx = min(i + BATCH_SIZE, len(hf_dataset))

    # Use .select() and .to_dict() for proper batch access
    batch = hf_dataset.select(range(i, end_idx)).to_dict()
    batch_results = convert_and_format(batch)

    # Only keep valid results
    converted_texts.extend(batch_results)
    dataset_idx = end_idx

    # Progress
    valid_count = len(converted_texts)
    print(f"  Processed {dataset_idx}/{len(hf_dataset)} samples, {valid_count} valid conversions", end="\r")

    # Save checkpoint and optimize memory
    if dataset_idx % SAVE_EVERY == 0:
        with open(CONVERT_CHECKPOINT, "w") as f:
            json.dump({
                "texts": converted_texts,
                "dataset_idx": dataset_idx,
            }, f)
        print(f"  Checkpoint saved at dataset_idx={dataset_idx}, {valid_count} valid samples")

        # Optimize GPU RAM
        gc.collect()
        torch.cuda.empty_cache()

# Save final checkpoint
with open(CONVERT_CHECKPOINT, "w") as f:
    json.dump({
        "texts": converted_texts,
        "dataset_idx": dataset_idx,

print(f"\nConversion complete: {len(converted_texts)} valid samples from {dataset_idx} total")
